            if host_to_rem in self.hosts:
                self.hosts.remove(host_to_rem)
        Database.remove_hosts(host_list_to_del)

    def add_admin(self, name: str, email: str, password: str, smtp_server: str, permission_level: int):
        new_admin = Admin(name, email, password, smtp_server, permission_level)
        self.admins.append(new_admin)
        Database.store_admins([new_admin])

    def remove_admin(self, admin_list_to_del: list[Admin]):
        for admin_to_del in admin_list_to_del:
            if admin_to_del in self.admins:
                self.admins.remove(admin_to_del)
        Database.remove_admins(admin_list_to_del)

    def get_test_collections(self) -> list[Test_Collection]:
        tc_list: list[Test_Collection] = []
        for host in self.hosts:
            rows = Database.get_tests_host_id(host.id)
            ping_list: list[Ping] = []
            for row in rows:
                p = Ping()
                p.timestamp = row.timestamp
                p.on = row.result
                ping_list.append(p)
            tc = Test_Collection(host, ping_list)
            tc_list.append(tc)
        return tc_list

    def collect_metrics(self):
        # Atualiza métricas para todos os hosts
        for tc in self.tests:
            tc.get_metrics()
        # Hosts sem Test_Collection ainda
        known_hosts = {tc.host.id for tc in self.tests}
        for host in self.hosts:
            if host.id not in known_hosts:
                tc = Test_Collection(host)
                tc.get_metrics()
                self.tests.append(tc)

    def update_Tests_Table(self):
        Database.store_tests(self.tests)

    def remove_tests(self, host_tests_to_del: list[Test_Collection]):
        for to_del in host_tests_to_del:
            if to_del in self.tests:
                self.tests.remove(to_del)
        Database.remove_tests_ping(host_tests_to_del)

    def alert_trigger(self):
        # Exemplo simples: se último ping de um host falhar, enviar email a todos os admins
        for tc in self.tests:
            if tc.test_metrics:
                last_ping = tc.test_metrics[-1]
                if not last_ping.on:
                    for admin in self.admins:
                        admin.send_email(tc.host.name)

    def get_status(self, host: Host) -> bool | None:
        for tc in self.tests:
            if tc.host.id == host.id and tc.test_metrics:
                return tc.test_metrics[-1].on
        return None

    def get_history(self, host: Host):
        for tc in self.tests:
            if tc.host.id == host.id:
                return tc.test_metrics
        return []

    def export_data(self, host: Host, filename: str = "export_host.txt"):
        history = self.get_history(host)
        with open(filename, "w", encoding="utf-8") as f:
            for ping in history:
                f.write(f"{ping.timestamp} - {'UP' if ping.on else 'DOWN'}\n")

    def _build_widgets(self):

        self.frame_info = tk.Frame(self.root)
        self.frame_info.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=5, pady=5)

        self.label_status = tk.Label(self.frame_info, text="Estado: -")
        self.label_status.pack(anchor="w")

        self.btn_show_history = tk.Button(self.frame_info, text="Mostrar Histórico", command=self.show_history)
        self.btn_show_history.pack(pady=2, anchor="w")

        self.btn_export = tk.Button(self.frame_info, text="Exportar Histórico", command=self.export_history)
        self.btn_export.pack(pady=2, anchor="w")

    def _load_hosts(self):
        self.listbox_hosts.delete(0, tk.END)
        for host in self.monitor_service.hosts:
            self.listbox_hosts.insert(tk.END, f"{host.id} - {host.name} ({host.ip})")

    def on_host_select(self, event):
        selection = self.listbox_hosts.curselection()
        if not selection:
            self.current_host = None
            self.label_status.config(text="Estado: -")
            return
        index = selection[0]
        self.current_host = self.monitor_service.hosts[index]
        status = self.monitor_service.get_status(self.current_host)
        if status is None:
            self.label_status.config(text="Estado: sem dados")
        else:
            self.label_status.config(text=f"Estado: {'UP' if status else 'DOWN'}")

    def add_host(self):
        ip = simpledialog.askstring("Novo Host", "IP:")
        if not ip:
            return
        name = simpledialog.askstring("Novo Host", "Nome:")
        if not name:
            return
        perm = simpledialog.askinteger("Novo Host", "Permission level:", minvalue=0, maxvalue=10)
        if perm is None:
            return
        new_id = max([h.id for h in self.monitor_service.hosts], default=0) + 1
        self.monitor_service.add_host(new_id, ip, name, perm)
        self._load_hosts()

    def remove_host(self):
        if not self.current_host:
            messagebox.showwarning("Remover Host", "Nenhum host selecionado.")
            return
        self.monitor_service.remove_host([self.current_host])
        self.current_host = None
        self._load_hosts()
        self.label_status.config(text="Estado: -")

    def refresh(self):
        self.monitor_service.collect_metrics()
        self.monitor_service.update_Tests_Table()
        if self.current_host:
            status = self.monitor_service.get_status(self.current_host)
            if status is None:
                self.label_status.config(text="Estado: sem dados")
            else:
                self.label_status.config(text=f"Estado: {'UP' if status else 'DOWN'}")

    def show_history(self):
        if not self.current_host:
            messagebox.showwarning("Histórico", "Nenhum host selecionado.")
            return
        history = self.monitor_service.get_history(self.current_host)
        win = tk.Toplevel(self.root)
        win.title(f"Histórico - {self.current_host.name}")
        text = tk.Text(win, width=60, height=20)
        text.pack()
        if not history:
            text.insert(tk.END, "Sem histórico.\n")
        else:
            for ping in history:
                text.insert(tk.END, f"{ping.timestamp} - {'UP' if ping.on else 'DOWN'}\n")

    def export_history(self):
        if not self.current_host:
            messagebox.showwarning("Exportar", "Nenhum host selecionado.")
            return
        filename = f"history_host_{self.current_host.id}.txt"
        self.monitor_service.export_data(self.current_host, filename)
        messagebox.showinfo("Exportar", f"Histórico exportado para {filename}")


if __name__ == "__main__":
    root = tk.Tk()
