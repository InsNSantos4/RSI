#!/bin/sh

PROG=$(basename "$0")

STATE=${HOME:-.}/.handin_mgr
TODOS=$STATE/todos.csv
RUNS=$STATE/runs
LOCKS=$STATE/locks

err() {
    printf '%s\n' "$*" >&2
}

die() {
    err "$*"
    exit 1
}

ensure_state() {
    [ -d "$STATE" ] || mkdir -p "$STATE" || die "Erro ao criar $STATE"
    [ -d "$RUNS" ]  || mkdir -p "$RUNS"  || die "Erro ao criar $RUNS"
    [ -d "$LOCKS" ] || mkdir -p "$LOCKS" || die "Erro ao criar $LOCKS"
    [ -f "$TODOS" ] || : >"$TODOS"       || die "Erro ao criar $TODOS"
}

acquire_lock() {
    lockdir="$LOCKS/main.lock"
    while ! mkdir "$lockdir" 2>/dev/null; do
        sleep 0.1
    done
}

release_lock() {
    lockdir="$LOCKS/main.lock"
    rmdir "$lockdir" 2>/dev/null || :
}

print_general_help() {
    cat <<EOF
Uso dentro da shell:
  handin_mgr todo-add "titulo" [-p P] [-d AAAA-MM-DD] [-t tag1,tag2]
  handin_mgr todo-list [-a] [-s due|prio|created] [-t tag] [-p minPrio]
  handin_mgr todo-done <id>
  handin_mgr todo-search <texto>
  handin_mgr handin-ingest <inbox_dir> <repo_dir> [-m]
  handin_mgr handin-check <repo_dir> [-o relatorio.csv]
  handin_mgr handin-summary <repo_dir> [-u UC] [-t TP#]

Comandos especiais da shell:
  help
  exit
EOF
}

cmd_todo_add() {
    [ "${1-}" = "-h" ] && { print_todo_add_help; return 0; }

    ensure_state

    [ $# -gt 0 ] || die "Falta titulo"
    title=$1
    shift

    prio=3
    due=
    tags=

    while [ $# -gt 0 ]; do
        case $1 in
            -p)
                shift
                [ $# -gt 0 ] || die "Falta valor para -p"
                is_prio "$1" || die "Prioridade invalida: $1"
                prio=$1
                ;;
            -d)
                shift
                [ $# -gt 0 ] || die "Falta valor para -d"
                is_iso_date "$1" || die "Data invalida (esperado AAAA-MM-DD): $1"
                due=$1
                ;;
            -t)
                shift
                [ $# -gt 0 ] || die "Falta valor para -t"
                tags=$1
                ;;
            *)
                die "Opcao desconhecida em todo-add: $1"
                ;;
        esac
        shift
    done

    acquire_lock

    id=$(next_id)
    created_at=$(now_iso)
    status=OPEN
    done_at=

    printf '%s;%s;%s;%s;%s;%s;%s;%s\n' \
        "$id" "$status" "$prio" "$due" "$tags" "$created_at" "$done_at" "$title" >>"$TODOS"

    release_lock

    printf 'Criada tarefa %s\n' "$id"
}

cmd_todo_list() {
    [ "${1-}" = "-h" ] && { print_todo_list_help; return 0; }

    show_all=0
    sort_key=created
    filter_tag=
    min_prio=1

    while [ $# -gt 0 ]; do
        case $1 in
            -a) show_all=1 ;;
            -s)
                shift
                case ${1-} in
                    due|prio|created) sort_key=$1 ;;
                    *) die "Chave de ordenacao invalida: $1" ;;
                esac
                ;;
            -t)
                shift
                filter_tag=${1-}
                ;;
            -p)
                shift
                is_prio "${1-}" || die "Prioridade invalida: $1"
                min_prio=$1
                ;;
            *)
                die "Opcao desconhecida em todo-list: $1"
                ;;
        esac
        shift
    done

    ensure_state
    [ -s "$TODOS" ] || { echo "Sem tarefas."; return 0; }

    case $sort_key in
        prio)    sort_cmd=" -t ';' -k3,3nr" ;;
        due)     sort_cmd=" -t ';' -k4,4" ;;
        created) sort_cmd=" -t ';' -k6,6" ;;
    esac

    awk -F';' -v show_all="$show_all" -v tag="$filter_tag" -v min_prio="$min_prio" '
        {
            id=$1; status=$2; prio=$3; due=$4; tags=$5; created=$6; done_at=$7; title=$8
            if (!show_all && status != "OPEN") next
            if (prio+0 < min_prio) next
            if (tag != "" && index(","tags",", ","tag",") == 0) next
            print $0
        }
    ' "$TODOS" | eval sort"$sort_cmd" | awk -F';' '
        {
            id=$1; status=$2; prio=$3; due=$4; tags=$5; created=$6; done_at=$7; title=$8
            if (due == "") due_str="-"
            else due_str=due
            printf "[%s] (P%s) %s %s tags:%s - %s\n", id, prio, due_str, status, tags, title
        }
    '
}

cmd_todo_done() {
    [ "${1-}" = "-h" ] && { print_todo_done_help; return 0; }

    [ $# -eq 1 ] || die "Uso: todo-done <id>"
    id_target=$1

    case $id_target in
        *[!0-9]*|"") die "ID invalido: $id_target" ;;
    esac

    ensure_state
    [ -s "$TODOS" ] || die "Sem tarefas."

    acquire_lock
    tmp=$(mktemp) || { release_lock; die "mktemp falhou"; }
    now=$(now_iso)
    found=0
    already=0

    awk -F';' -v idt="$id_target" -v now="$now" '
        BEGIN { OFS=";" }
        {
            if ($1 == idt) {
                found=1
                if ($2 == "DONE") {
                    already=1
                } else {
                    $2="DONE"
                    $7=now
                }
            }
            print $0
        }
        END {
            if (!found) exit 2
            if (already) exit 3
        }
    ' "$TODOS" >"$tmp"
    rc=$?

    if [ $rc -eq 2 ]; then
        rm -f "$tmp"
        release_lock
        die "ID nao encontrado: $id_target"
    fi

    mv "$tmp" "$TODOS"
    release_lock

    if [ $rc -eq 3 ]; then
        echo "Tarefa $id_target ja estava DONE (mantido timestamp)."
    else
        echo "Tarefa $id_target marcada como DONE"
    fi
}

cmd_todo_search() {
    [ "${1-}" = "-h" ] && { print_todo_search_help; return 0; }

    [ $# -ge 1 ] || die "Uso: todo-search <texto>"
    text=$*

    ensure_state
    [ -s "$TODOS" ] || { echo "Sem tarefas."; return 0; }

    awk -F';' -v q="$text" '
        BEGIN { IGNORECASE=1 }
        {
            id=$1; status=$2; prio=$3; due=$4; tags=$5; created=$6; done_at=$7; title=$8
            if (index(title, q) > 0) {
                if (due == "") due_str="-"
                else due_str=due
                printf "[%s] (P%s) %s %s tags:%s - %s\n", id, prio, due_str, status, tags, title
            }
        }
    ' "$TODOS"
}

valid_name() {
    # ALUNO_UC_TP#_YYYYMMDDHHMM
    case $1 in
        *_*_TP[0-9]_[0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9][0-9]) return 0 ;;
        *) return 1 ;;
    esac
}

split_name() {
    base=$1
    ALUNO=${base%%_*}
    rest=${base#*_}
    UC=${rest%%_*}
    rest=${rest#*_}
    TP=${rest%%_*}
    TS=${rest#*_}
    echo "$ALUNO;$UC;$TP;$TS"
}

cmd_handin_ingest() {
    [ "${1-}" = "-h" ] && { print_handin_ingest_help; return 0; }

    [ $# -ge 2 ] || die "Uso: handin-ingest <inbox_dir> <repo_dir> [-m]"
    inbox=$1
    repo=$2
    shift 2
    move=0
    [ "${1-}" = "-m" ] && move=1

    [ -d "$inbox" ] || die "inbox_dir invalido"
    [ -d "$repo" ]  || die "repo_dir invalido"

    ensure_state
    ts=$(date +"%Y-%m-%d_%H%M%S")
    report="$RUNS/ingest_$ts.txt"

    find "$inbox" -mindepth 1 -maxdepth 1 | while IFS= read -r path; do
        name=$(basename "$path")
        base=$name
        case $name in
            *.zip|*.tar|*.tar.gz)
                base=${name%%.*}
                ;;
        esac

        if ! valid_name "$base"; then
            echo "FAIL;$path;nome_invalido" >>"$report"
            continue
        fi

        info=$(split_name "$base")
        ALUNO=$(echo "$info" | cut -d';' -f1)
        UC=$(echo    "$info" | cut -d';' -f2)
        TP=$(echo    "$info" | cut -d';' -f3)
        TS=$(echo    "$info" | cut -d';' -f4)

        dest="$repo/$UC/$TP/$ALUNO/$TS"
        mkdir -p "$dest" || { echo "FAIL;$path;mkdir_dest" >>"$report"; continue; }

        if [ -d "$path" ]; then
            if [ $move -eq 1 ]; then
                mv "$path" "$dest/" 2>/dev/null || { echo "FAIL;$path;mv_dir" >>"$report"; continue; }
            else
                cp -R "$path" "$dest/" 2>/dev/null || { echo "FAIL;$path;cp_dir" >>"$report"; continue; }
            fi
            echo "OK;$path;$dest" >>"$report"
            continue
        fi

        case $name in
            *.zip)
                if command -v unzip >/dev/null 2>&1; then
                    cp "$path" "$dest/" || { echo "FAIL;$path;cp_zip" >>"$report"; continue; }
                    (cd "$dest" && unzip -q "$name" && rm -f "$name") || { echo "FAIL;$path;unzip" >>"$report"; continue; }
                    [ $move -eq 1 ] && rm -f "$path"
                    echo "OK;$path;$dest" >>"$report"
                else
                    echo "FAIL;$path;sem_unzip" >>"$report"
                fi
                ;;
            *.tar|*.tar.gz)
                if command -v tar >/dev/null 2>&1; then
                    cp "$path" "$dest/" || { echo "FAIL;$path;cp_tar" >>"$report"; continue; }
                    (cd "$dest" && tar xf "$name" && rm -f "$name") || { echo "FAIL;$path;tar" >>"$report"; continue; }
                    [ $move -eq 1 ] && rm -f "$path"
                    echo "OK;$path;$dest" >>"$report"
                else
                    echo "FAIL;$path;sem_tar" >>"$report"
                fi
                ;;
            *)
                echo "FAIL;$path;tipo_desconhecido" >>"$report"
                ;;
        esac
    done
}

check_one_delivery() {
    root=$1
    rel=$2

    base=$(basename "$root")
    info=$(split_name "$base")
    ALUNO=$(echo "$info" | cut -d';' -f1)
    UC=$(echo    "$info" | cut -d';' -f2)
    TP=$(echo    "$info" | cut -d';' -f3)
    TS=$(echo    "$info" | cut -d';' -f4)

    status=OK
    problems=""

    # README
    if [ ! -f "$root/README.md" ] && [ ! -f "$root/README.txt" ]; then
        status=FAIL
        problems="${problems:+$problems,}missing_readme"
    fi

    # src/
    if [ ! -d "$root/src" ]; then
        status=FAIL
        problems="${problems:+$problems,}missing_src"
    else
        src_count=$(find "$root/src" -type f | wc -l)
        [ "$src_count" -ge 1 ] || {
            status=FAIL
            problems="${problems:+$problems,}missing_src"
        }
    fi

    # proibidos
    if find "$root" -type d \( -name node_modules -o -name dist -o -name build \) | grep -q .; then
        status=FAIL
        problems="${problems:+$problems,}forbidden_dir"
    fi

    too_large=0
    binary_found=0
    total_files=0
    num_src_files=0
    total_lines=0

    find "$root" -type f | while IFS= read -r f; do
        total_files=$((total_files+1))
        size=$(stat -c %s "$f" 2>/dev/null || echo 0)
        if [ "$size" -gt $((5*1024*1024)) ]; then
            too_large=1
        fi
        if ! grep -Iq . "$f" 2>/dev/null; then
            binary_found=1
        fi
        case $f in
            "$root"/src/*)
                num_src_files=$((num_src_files+1))
                if grep -Iq . "$f" 2>/dev/null; then
                    lines=$(wc -l <"$f" 2>/dev/null || echo 0)
                    total_lines=$((total_lines+lines))
                fi
                ;;
        esac
    done

    if [ $too_large -eq 1 ]; then
        status=FAIL
        problems="${problems:+$problems,}too_large"
    fi
    if [ $binary_found -eq 1 ]; then
        status=FAIL
        problems="${problems:+$problems,}binary_found"
    fi

    printf '%s;%s;%s;%s;%s;%s;%s;%s;%s\n' \
        "$UC" "$TP" "$ALUNO" "$TS" "$status" "$num_src_files" "$total_files" "$total_lines" "$problems"

    if [ "$status" = "FAIL" ]; then
        reason=$problems
        prio=4
        case $problems in
            *missing_src*|*binary_found*) prio=5 ;;
        esac
        ensure_state
        acquire_lock
        id=$(next_id)
        created_at=$(now_iso)
        title="Corrigir entrega $ALUNO $UC $TP ($reason)"
        tags="handin,$UC,$TP"
        printf '%s;%s;%s;%s;%s;%s;%s;%s\n' \
            "$id" "OPEN" "$prio" "" "$tags" "$created_at" "" "$title" >>"$TODOS"
        release_lock
    fi
}

cmd_handin_check() {
    [ "${1-}" = "-h" ] && { print_handin_check_help; return 0; }

    [ $# -ge 1 ] || die "Uso: handin-check <repo_dir> [-o relatorio]"
    repo=$1
    shift
    out=
    if [ "${1-}" = "-o" ]; then
        shift
        [ $# -ge 1 ] || die "Falta nome de ficheiro para -o"
        out=$1
    fi

    [ -d "$repo" ] || die "repo_dir invalido"

    tmp=$(mktemp) || die "mktemp falhou"
    find "$repo" -mindepth 4 -maxdepth 4 -type d | while IFS= read -r d; do
        base=$(basename "$d")
        valid_name "$base" || continue
        check_one_delivery "$d" "$repo"
    done >"$tmp"

    if [ -n "$out" ]; then
        mv "$tmp" "$out"
    else
        cat "$tmp"
        rm -f "$tmp"
    fi
}

cmd_handin_summary() {
    [ "${1-}" = "-h" ] && { print_handin_summary_help; return 0; }

    [ $# -ge 1 ] || die "Uso: handin-summary <repo_dir> [-u UC] [-t TP#]"
    repo=$1
    shift
    filter_uc=
    filter_tp=

    while [ $# -gt 0 ]; do
        case $1 in
            -u) shift; filter_uc=${1-} ;;
            -t) shift; filter_tp=${1-} ;;
            *) die "Opcao desconhecida em handin-summary: $1" ;;
        esac
        shift
    done

    [ -d "$repo" ] || die "repo_dir invalido"

    tmp=$(mktemp) || die "mktemp falhou"
    cmd_handin_check "$repo" >"$tmp"

    awk -F';' -v u="$filter_uc" -v t="$filter_tp" '
        BEGIN {
            total=0; ok=0; fail=0;
        }
        {
            UC=$1; TP=$2; ALUNO=$3; TS=$4; STATUS=$5; ns=$6; tf=$7; tl=$8; probs=$9;
            if (u != "" && UC != u) next
            if (t != "" && TP != t) next
            total++
            if (STATUS=="OK") ok++; else fail++;
            key=UC" "TP" "ALUNO" "TS
            files[key]=tf
            lines[key]=tl
            if (STATUS=="FAIL") alunos[ALUNO]=1
        }
        END {
            printf "Resumo de entregas\n"
            if (u!="") printf "UC: %s\n", u
            if (t!="") printf "TP: %s\n", t
            printf "Total: %d\nOK: %d\nFAIL: %d\n\n", total, ok, fail

            printf "Alunos com pelo menos uma FAIL:\n"
            for (a in alunos) printf "  %s\n", a

            # tops simples (sem sort sofisticado)
            printf "\n(Top 5 por total_files nao implementado com sort real nesta versao simples)\n"
        }
    ' "$tmp"

    rm -f "$tmp"
}

dispatch() {
    [ $# -gt 0 ] || { err "Falta comando"; return 1; }

    case $1 in
        todo-add)       shift; cmd_todo_add "$@" ;;
        todo-list)      shift; cmd_todo_list "$@" ;;
        todo-done)      shift; cmd_todo_done "$@" ;;
        todo-search)    shift; cmd_todo_search "$@" ;;
        handin-ingest)  shift; cmd_handin_ingest "$@" ;;
        handin-check)   shift; cmd_handin_check "$@" ;;
        handin-summary) shift; cmd_handin_summary "$@" ;;
        -h|--help)      print_general_help ;;
        *)              err "Comando invalido: $1"; return 1 ;;
    esac
}

run_shell() {
    ensure_state

    printf 'Shell interativa de %s\n' "$PROG"
    printf 'Escreve help para ajuda e exit para sair\n'

    while :; do
        printf 'shell> '
        IFS= read -r line || { printf '\n'; break; }

        case $line in
            '') continue ;;
            exit|exist) break ;;
            help) print_general_help; continue ;;
            "$PROG "*) cmdline=${line#"$PROG "} ;;
            "$PROG") err "Falta o comando"; continue ;;
            *) err "Usa comandos no formato: $PROG <comando> ..."; continue ;;
        esac

        set -- $cmdline
        [ $# -gt 0 ] || continue

        dispatch "$@"
        rc=$?
        [ "$rc" -eq 0 ] || err "Comando terminou com codigo $rc"
    done
}

# handin_mgr — Gestão de tarefas e entregas

## Descrição

O `handin_mgr` é um script POSIX (`/bin/sh`) que implementa:

- Gestão de tarefas (to-do) persistente em `~/.handin_mgr/todos.csv`
- Gestão de entregas de trabalhos (hand-ins), com:
  - ingestão de entregas
  - validação de estrutura e conteúdos
  - geração de relatórios
  - criação automática de tarefas em caso de falha

Todo o estado é mantido em:

- `~/.handin_mgr/todos.csv`
- `~/.handin_mgr/runs/`
- `~/.handin_mgr/locks/`

---

## Execução

Tornar o ficheiro executável:

```sh
chmod +x handin_mgr
./handin_mgr

O programa entra em modo shell interativo.
Os comandos devem ser escritos no formato:

handin_mgr <comando> [opcoes] [args]

Comandos especiais da shell:

    Gera resumo agregado das entregas (filtrado por UC e/ou TP):

        total de entregas

    Nesta versão, os “top 5” são mencionados mas não totalmente ordenados com sort avançado (limitação documentada).

Limitações e suposições

    Assume date -d disponível (Linux).

    Assume stat -c (GNU coreutils).

    Se unzip ou tar não existirem, as entregas comprimidas são marcadas como FAIL com motivo apropriado.

    A shell interativa faz parsing simples por espaços (set -- $cmdline), não trata aspas complexas como uma shell real.

Ferramentas opcionais

    unzip — para extrair .zip

    tar — para extrair .tar e .tar.gz

Se não existirem, o script:

    não extrai os ficheiros

    marca as entregas como FAIL com motivo sem_unzip ou sem_tar.
